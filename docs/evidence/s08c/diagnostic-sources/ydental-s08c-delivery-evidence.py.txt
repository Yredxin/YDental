# Mask actual credentials from private access/config material without printing them.
secrets=set()
def collect(v,k=''):
 if isinstance(v,dict):
  for key,value in v.items():collect(value,key)
 elif isinstance(v,list):
  for x in v:collect(x,k)
 elif isinstance(v,str) and len(v)>=12 and re.search(r'password|secret|token|sid|session_key|cookie',k,re.I):secrets.add(v)
for run in (ROOT/'.local').glob('ydental_s08c_test_*'):

 for p in run.glob('*access*.json*'):
  try:collect(json.loads(p.read_text()))
  except (ValueError,UnicodeError):pass

 if conf.exists():
  for line in conf.read_text().splitlines():
   if re.match(r'\s*(db_password|admin_passwd)\s*=',line):
    value=line.split('=',1)[1].strip()
    if len(value)>=12:secrets.add(value)
